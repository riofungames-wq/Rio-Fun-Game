// ====================================== // RIO MAGGI POINT // PREMIUM CUSTOMER REVIEW SYSTEM // FEEDBACK.JS // FINAL CLEAN VERSION // ====================================== // ====================================== // FIREBASE CENTRAL IMPORT // ====================================== import { auth, db } from "./firebase.js"; // ====================================== // FIREBASE AUTH // ====================================== import { onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js"; // ====================================== // FIREBASE FIRESTORE // ====================================== import { doc, getDoc, addDoc, collection, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js"; // ====================================== // DOM ELEMENTS // ====================================== const feedbackPhoto = document.getElementById("feedbackPhoto"); const feedbackName = document.getElementById("feedbackName"); const feedbackMember = document.getElementById("feedbackMember"); const stars = document.querySelectorAll(".star"); const ratingText = document.getElementById("ratingText"); const emojiButtons = document.querySelectorAll(".emoji-btn"); const reviewText = document.getElementById("reviewText"); const charCount = document.getElementById("charCount"); const submitFeedback = document.getElementById("submitFeedback"); const thankYouBox = document.getElementById("thankYouBox"); const feedbackError = document.getElementById("feedbackError"); const submitBtnText = submitFeedback ? submitFeedback.querySelector(".submit-btn-text") : null; const submitLoadingIcon = submitFeedback ? submitFeedback.querySelector(".submit-loading-icon") : null; // ====================================== // REVIEW STATE // ====================================== let selectedRating = 0; let selectedEmoji = ""; let currentUser = null; let isSubmitting = false; // ====================================== // RATING LABELS // ====================================== const ratingLabels = { 1: "Very Bad", 2: "Not Good", 3: "Average", 4: "Very Good", 5: "Excellent" }; // ====================================== // SHOW ERROR // ====================================== function showFeedbackError(message) { if (!feedbackError) { alert(message); return; } feedbackError.textContent = message; feedbackError.style.display = "block"; } // ====================================== // HIDE ERROR // ====================================== function hideFeedbackError() { if (!feedbackError) { return; } feedbackError.textContent = ""; feedbackError.style.display = "none"; } // ====================================== // LOAD CUSTOMER PROFILE // ====================================== async function loadCustomerProfile(user) { try { const userRef = doc( db, "users", user.uid ); const userSnap = await getDoc(userRef); if (!userSnap.exists()) { console.warn( "Customer profile not found." ); if (feedbackName) { feedbackName.textContent = user.displayName || "Customer"; } if (feedbackMember) { feedbackMember.textContent = "RIO-000000"; } if ( feedbackPhoto && user.photoURL ) { feedbackPhoto.src = user.photoURL; } return; } const data = userSnap.data(); // ================================== // CUSTOMER NAME // ================================== if (feedbackName) { feedbackName.textContent = data.name || user.displayName || "Customer"; } // ================================== // MEMBER ID // ================================== if (feedbackMember) { feedbackMember.textContent = data.memberId || "RIO-000000"; } // ================================== // CUSTOMER PHOTO // ================================== if (feedbackPhoto) { const photo = data.photo || data.photoURL || user.photoURL; if (photo) { feedbackPhoto.src = photo; } } } catch (error) { console.error( "Failed to load customer profile:", error ); if (feedbackName) { feedbackName.textContent = user.displayName || "Customer"; } } } // ====================================== // SET RATING // ====================================== function setRating(rating) { selectedRating = Number(rating); // ================================== // UPDATE STARS // ================================== stars.forEach( (star) => { const starRating = Number( star.dataset.rate ); if ( starRating <= selectedRating ) { star.classList.remove( "fa-regular" ); star.classList.add( "fa-solid" ); star.classList.add( "active" ); star.setAttribute( "aria-checked", "true" ); } else { star.classList.remove( "fa-solid" ); star.classList.remove( "active" ); star.classList.add( "fa-regular" ); star.setAttribute( "aria-checked", "false" ); } } ); // ================================== // UPDATE RATING TEXT // ================================== if (ratingText) { ratingText.textContent = ratingLabels[ selectedRating ] || "Tap A Star"; } } // ====================================== // RESET RATING // ====================================== function resetRating() { selectedRating = 0; stars.forEach( (star) => { star.classList.remove( "fa-solid" ); star.classList.remove( "active" ); star.classList.add( "fa-regular" ); star.setAttribute( "aria-checked", "false" ); } ); if (ratingText) { ratingText.textContent = "Tap A Star"; } } // ====================================== // SELECT EMOJI // ====================================== function selectEmoji(button) { emojiButtons.forEach( (item) => { item.classList.remove( "active" ); item.setAttribute( "aria-pressed", "false" ); } ); button.classList.add( "active" ); button.setAttribute( "aria-pressed", "true" ); selectedEmoji = button.dataset.emotion || button.textContent.trim(); } // ====================================== // RESET EMOJI // ====================================== function resetEmoji() { selectedEmoji = ""; emojiButtons.forEach( (button) => { button.classList.remove( "active" ); button.setAttribute( "aria-pressed", "false" ); } ); } // ====================================== // UPDATE CHARACTER COUNT // ====================================== function updateCharacterCount() { if ( !reviewText || !charCount ) { return; } if ( reviewText.value.length > 300 ) { reviewText.value = reviewText.value.substring( 0, 300 ); } charCount.textContent = reviewText.value.length; } // ====================================== // SET SUBMIT LOADING STATE // ====================================== function setSubmitLoading(loading) { if (!submitFeedback) { return; } submitFeedback.disabled = loading; if (submitBtnText) { submitBtnText.textContent = loading ? "Submitting..." : "Submit Feedback"; } if (submitLoadingIcon) { submitLoadingIcon.style.display = loading ? "inline-block" : "none"; } } // ====================================== // SHOW SUCCESS MESSAGE // ====================================== function showThankYou() { if (!thankYouBox) { return; } thankYouBox.style.display = "block"; thankYouBox.scrollIntoView({ behavior: "smooth", block: "center" }); setTimeout( () => { if (thankYouBox) { thankYouBox.style.display = "none"; } }, 5000 ); } // ====================================== // RESET FEEDBACK FORM // ====================================== function resetFeedbackForm() { if (reviewText) { reviewText.value = ""; } if (charCount) { charCount.textContent = "0"; } resetRating(); resetEmoji(); } // ====================================== // SUBMIT FEEDBACK // ====================================== async function handleFeedbackSubmit() { // ================================== // PREVENT DOUBLE SUBMISSION // ================================== if (isSubmitting) { return; } // ================================== // CHECK AUTH // ================================== if (!currentUser) { showFeedbackError( "Please login first." ); return; } // ================================== // CHECK RATING // ================================== if ( selectedRating < 1 ) { showFeedbackError( "Please select a star rating." ); return; } // ================================== // GET REVIEW // ================================== const review = reviewText ? reviewText.value.trim() : ""; // ================================== // HIDE OLD ERROR // ================================== hideFeedbackError(); // ================================== // LOCK SUBMISSION // ================================== isSubmitting = true; setSubmitLoading(true); try { // ================================== // SAVE TO FIRESTORE // ================================== await addDoc( collection( db, "feedback" ), { userId: currentUser.uid, rating: selectedRating, emoji: selectedEmoji, review: review, createdAt: serverTimestamp() } ); // ================================== // SUCCESS // ================================== console.log( "Feedback submitted successfully." ); showThankYou(); resetFeedbackForm(); } catch (error) { console.error( "Feedback submission failed:", error ); showFeedbackError( "Unable to submit feedback. Please try again." ); } finally { isSubmitting = false; setSubmitLoading(false); } } // ====================================== // STAR CLICK EVENTS // ====================================== stars.forEach( (star) => { star.addEventListener( "click", () => { setRating( star.dataset.rate ); } ); // ================================== // KEYBOARD SUPPORT // ================================== star.addEventListener( "keydown", (event) => { if ( event.key === "Enter" || event.key === " " ) { event.preventDefault(); setRating( star.dataset.rate ); } } ); } ); // ====================================== // EMOJI CLICK EVENTS // ====================================== emojiButtons.forEach( (button) => { button.addEventListener( "click", () => { selectEmoji( button ); } ); } ); // ====================================== // REVIEW INPUT // ====================================== if (reviewText) { reviewText.addEventListener( "input", updateCharacterCount ); updateCharacterCount(); } // ====================================== // SUBMIT BUTTON // ====================================== if (submitFeedback) { submitFeedback.addEventListener( "click", handleFeedbackSubmit ); } // ====================================== // AUTH STATE // ====================================== onAuthStateChanged( auth, async (user) => { if (!user) { currentUser = null; console.warn( "No logged-in user found." ); window.location.href = "login.html"; return; } currentUser = user; await loadCustomerProfile( user ); console.log( "Feedback page loaded for:", user.uid ); } ); // ====================================== // PREVENT IMAGE DRAGGING // ====================================== document .querySelectorAll("img") .forEach( (image) => { image.addEventListener( "dragstart", (event) => { event.preventDefault(); } ); } ); // ====================================== // PAGE READY // ====================================== document.addEventListener( "DOMContentLoaded", () => { document.body.classList.add( "feedback-page-ready" ); } ); // ====================================== // FINAL READY LOG // ====================================== console.log( "================================" ); console.log( "RIO MAGGI POINT" ); console.log( "Premium Customer Review System Ready" ); console.log( "Central Firebase Connected" ); console.log( "================================" );
